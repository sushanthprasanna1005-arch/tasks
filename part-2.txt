USE SCHEMA banking_db.raw;

CREATE OR REPLACE PROCEDURE banking_db.raw.proc_load_merge_customers()
RETURNS STRING
LANGUAGE SQL
AS
$$
BEGIN

  COPY INTO banking_db.raw.stg_customers
  FROM @banking_db.raw.gcs_stage/customers/
  FILE_FORMAT = (FORMAT_NAME = 'banking_db.raw.csv_ff')
  MATCH_BY_COLUMN_NAME = CASE_INSENSITIVE
  ON_ERROR = CONTINUE;

  MERGE INTO banking_db.core.dim_customer AS tgt
  USING (
    SELECT
      customer_id,
      customer_name,
      email,
      phone,
      country,
      city,
      TRY_TO_TIMESTAMP_NTZ(created_at) AS created_at
    FROM banking_db.raw.stg_customers
    WHERE customer_id IS NOT NULL
  ) AS src
  ON tgt.customer_id = src.customer_id
  WHEN MATCHED THEN
    UPDATE SET
      tgt.customer_name = src.customer_name,
      tgt.email         = src.email,
      tgt.phone         = src.phone,
      tgt.country       = src.country,
      tgt.city          = src.city,
      tgt.created_at    = COALESCE(tgt.created_at, src.created_at)
  WHEN NOT MATCHED THEN
    INSERT (customer_id, customer_name, email, phone, country, city, created_at)
    VALUES (src.customer_id, src.customer_name, src.email, src.phone,
            src.country, src.city, src.created_at);

  TRUNCATE TABLE banking_db.raw.stg_customers;

  RETURN 'Customers load + merge completed';
END;
$$;

CREATE OR REPLACE PROCEDURE banking_db.raw.proc_load_merge_transactions()
RETURNS STRING
LANGUAGE SQL
AS
$$
BEGIN

  COPY INTO banking_db.raw.stg_transactions
  FROM @banking_db.raw.gcs_stage/transactions/
  FILE_FORMAT = (FORMAT_NAME = 'banking_db.raw.csv_ff')
  MATCH_BY_COLUMN_NAME = CASE_INSENSITIVE
  ON_ERROR = CONTINUE;

  MERGE INTO banking_db.core.fact_transactions AS tgt
  USING (
    SELECT
      transaction_id,
      account_id,
      TRY_TO_TIMESTAMP_NTZ(transaction_date) AS transaction_date,
      transaction_type,
      TRY_TO_NUMBER(amount) AS amount,
      currency,
      channel,
      description
    FROM banking_db.raw.stg_transactions
    WHERE transaction_id IS NOT NULL
  ) AS src
  ON tgt.transaction_id = src.transaction_id
  WHEN MATCHED THEN

      tgt.account_id       = src.account_id,
      tgt.transaction_date = src.transaction_date,
      tgt.transaction_type = src.transaction_type,
      tgt.amount           = src.amount,
      tgt.currency         = src.currency,
      tgt.channel          = src.channel,
      tgt.description      = src.description
  WHEN NOT MATCHED THEN
    INSERT (transaction_id, account_id, transaction_date, transaction_type,
           amount, currency, channel, description)
    VALUES (src.transaction_id, src.account_id, src.transaction_date,
            src.transaction_type, src.amount, src.currency, src.channel, src.description);

  TRUNCATE TABLE banking_db.raw.stg_transactions;

  RETURN 'Transactions load + merge completed';
END;
$$;
